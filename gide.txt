# 1. Посмотреть что изменилось
git status

# 2. Увидеть детали изменений
git diff

# 3. Добавить все изменения в staging
git add .

# 4. Закоммитить с осмысленным сообщением
git commit -m "feat: добавил калькулятор в мобильный интерфейс"

# 5. Отправить в облако
git push origin main

git add .
git commit -m "fix: исправил баг с одиночными товарами"
git push


🌿 Работа с ветками (эксперименты без страха)
Зачем нужны ветки:
main — стабильная рабочая версия (никогда не ломаем)
feature/* — новые функции (экспериментируем)
fix/* — исправление багов
hotfix/* — срочные фиксы в production


# Переключаемся на main и обновляем
git checkout main
git pull origin main

# Создаём новую ветку
git checkout -b feature/new-calculator

# Работаем, коммитим сколько угодно раз
git add .
git commit -m "feat: добавлена кнопка калькулятора"
git add .
git commit -m "feat: калькулятор применяет результат к полю"

# Когда готово — отправляем в облако
git push -u origin feature/new-calculator


git checkout main
